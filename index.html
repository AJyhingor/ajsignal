// Pure JS RSI Calculation (Wilder's Smoothing) function calculateRSI(prices, period = 14) { if (prices.length <= period) return []; let rsi = []; let gains = []; let losses = []; // Calculate initial gains and losses for (let i = 1; i < prices.length; i++) { let difference = prices[i] - prices[i - 1]; gains.push(difference > 0 ? difference : 0); losses.push(difference < 0 ? Math.abs(difference) : 0); } // Calculate first average gain and loss let avgGain = gains.slice(0, period).reduce((a, b) => a + b, 0) / period; let avgLoss = losses.slice(0, period).reduce((a, b) => a + b, 0) / period; rsi.push(100 - (100 / (1 + (avgGain / (avgLoss || 1))))); // Apply Wilder's smoothing technique for the remaining values for (let i = period; i < gains.length; i++) { avgGain = ((avgGain * (period - 1)) + gains[i]) / period; avgLoss = ((avgLoss * (period - 1)) + losses[i]) / period; rsi.push(100 - (100 / (1 + (avgGain / (avgLoss || 1))))); } return rsi; }
